Arsist Engine

Arsist は AR グラス向けのクロスプラットフォーム開発エンジンです。
Electron と React で作られたエディタ上で、シーン・UI・ロジックを統合的に編集し、Unity バッチビルドを通じてデバイス向け APK を自動生成します。

Arsist is a cross-platform development engine for AR glasses.
Built with Electron and React, the editor provides integrated editing of scenes, UI, and logic, generating device-ready APKs through Unity batch builds.

コード

Code

var API_KEY = 'YOUR_API_KEY';
var CITY = 'Tokyo';

var url = 'https://api.openweathermap.org/data/2.5/weather?q=' + CITY
        + '&appid=' + API_KEY + '&units=metric&lang=ja';

api.get(url, function(res) {
  if (res === null) {
    ui.setText('weatherDesc', '取得失敗');
    return;
  }
  var data = JSON.parse(res);

  var tempC = Math.round(data.main.temp);
  var desc  = data.weather[0].description;
  var city  = data.name;

  ui.setText('cityName',     city);
  ui.setText('temperature',  tempC + '°C');
  ui.setText('weatherDesc',  desc);

  log('天気更新: ' + city + ' ' + tempC + '°C ' + desc);
});

必要なもの

Prerequisites

  • Discord Bot トークン（Discord Developer Portal で取得）
  • Discord Bot token (from Discord Developer Portal)
  • 対象チャンネルの ID
  • Target channel ID

Step 1: Python ブリッジサーバーを起動

Step 1: Start the Python bridge server

以下のスクリプトを discord_bridge.py として保存して実行します。

Save the following script as discord_bridge.py and run it.

#!/usr/bin/env python3
import os, sys, threading, socket
from collections import deque
from flask import Flask, jsonify
import discord

DISCORD_TOKEN = os.environ['DISCORD_BOT_TOKEN']
CHANNEL_ID    = int(os.environ['DISCORD_CHANNEL_ID'])
HTTP_PORT     = int(os.environ.get('DISCORD_BRIDGE_PORT', '8781'))
HOST_IP       = socket.gethostbyname(socket.gethostname())

messages = deque(maxlen=10)

class DiscordBot(discord.Client):
    async def on_ready(self):
        print(f'[Discord] Logged in as {self.user}')

    async def on_message(self, msg):
        if msg.author.bot or msg.channel.id != CHANNEL_ID:
            return
        messages.appendleft({
            'id': str(msg.id),
            'author': msg.author.display_name,
            'content': msg.content[:100],
            'ts': msg.created_at.isoformat(),
        })

app = Flask(__name__)

@app.get('/messages')
def get_messages():
    return jsonify({'items': list(messages), 'count': len(messages)})

def run_http():
    print(f'[Init] Starting HTTP server on {HOST_IP}:{HTTP_PORT}')
    app.run(host='0.0.0.0', port=HTTP_PORT, debug=False, use_reloader=False)

if __name__ == '__main__':
    threading.Thread(target=run_http, daemon=True).start()
    intents = discord.Intents.default()
    intents.message_content = True
    DiscordBot(intents=intents).run(DISCORD_TOKEN)

// *** endpoint を Python が出力したIPに変更してください ***
var endpoint = 'http://192.168.0.64:8781/messages';

if (!store.get('discord_initialized')) {
  ui.setText('discord_feed_text', '初期化中...');
  store.set('discord_initialized', true);
}

api.get(endpoint, function(res) {
  if (res === null) {
    ui.setText('discord_feed_text', 'サーバー接続失敗!\nendpoint: ' + endpoint);
    return;
  }

  var data;
  try { data = JSON.parse(res); }
  catch (e) { ui.setText('discord_feed_text', 'JSON解析エラー: ' + e); return; }

  if (data.error) {
    ui.setText('discord_feed_text', 'サーバーエラー: ' + data.error);
    return;
  }

  var items = data.items || [];
  if (items.length === 0) {
    ui.setText('discord_feed_text', 'メッセージなし');
    return;
  }

  var currentJson = JSON.stringify(items.map(function(x) { return x.id; }));
  if (currentJson === store.get('discord_last_ids')) return;

  var lines = [];
  for (var i = 0; i < items.length; i++) {
    lines.push(items[i].author + ': ' + items[i].content);
  }
  ui.setText('discord_feed_text', lines.join('\n'));
  store.set('discord_last_ids', currentJson);
  log('[Discord] Updated ' + items.length + ' messages');
});
